# ==================== MOTOR DIAGNOSTICS SERVICES ====================

-------------------  ros2 service call /reset_driver std_srvs/srv/Trigger && ros2 service call /initiate_motor std_srvs/srv/Trigger


# 1. Reset drivers (clear faults and reinitialize)
ros2 service call /reset_driver std_srvs/srv/Trigger

# 2. Initiate motors (manual re-initialization)
ros2 service call /initiate_motor std_srvs/srv/Trigger

# 3. Get fault information (read error codes)
ros2 service call /get_fault_info std_srvs/srv/Trigger

# 4. Get motor temperatures and bus voltage
ros2 service call /get_motor_temperature std_srvs/srv/Trigger

# 5. Halt motors (emergency stop)
ros2 service call /halt_motor std_srvs/srv/SetBool "{data: true}"

# 6. Resume motors (after halt)
ros2 service call /halt_motor std_srvs/srv/SetBool "{data: false}"

# 7. Enable park mode (motors hold position when stopped)
ros2 service call /set_park_mode std_srvs/srv/SetBool "{data: true}"

# 8. Disable park mode
ros2 service call /set_park_mode std_srvs/srv/SetBool "{data: false}"
